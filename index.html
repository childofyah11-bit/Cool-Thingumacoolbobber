${result !== undefined ? result : 'Statement executed successfully.'}`, 'success'); } catch (err) { appendLog(` > RUNTIME EXCEPTION:
${err.message}`, 'fail'); } return; } const input = fullInput.toLowerCase(); switch(input) { case 'help': appendLog(" > System commands: [status, neofetch, clear, matrix, ping]

The Matrix has you. 🖥️", "success"); break; case 'ping': appendLog(` > Manual intercept checked: ${latencyText.innerText}`, "info"); break; case 'neofetch': appendLog(" 🖥️ OS: Matrix Runtime Engine v6.0
